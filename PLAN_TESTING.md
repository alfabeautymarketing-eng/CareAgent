## 📌 ОБЗОР ПРОЕКТА

### Цель
Создать профессиональную систему логирования с визуальным дашбордом для полного контроля над всеми операциями в системе AgentCare. Это позволит быстро тестировать каждую кнопку меню и видеть детальный результат выполнения.

### Текущая проблема
- Листы "Журнал синхро" и "Журнал логов" заполняются автоматически, но сложно анализировать
- Нет визуального интерфейса для просмотра логов в реальном времени
- Сложно отследить результат выполнения конкретной функции
- Меню структурировано не интуитивно - логи и операции разбросаны по разным местам

### Результат после реализации
- ✅ Единая точка доступа ко всем логам - **Дашборд**
- ✅ Красивый интерфейс с фильтрами и поиском
- ✅ Реструктурированное меню: Экосистема → Логи → Агент → остальное
- ✅ Подробное логирование каждой функции с входными/выходными данными
- ✅ Возможность быстро найти и зафиксировать ошибку

## 🎯 ЭТАП 1: АРХИТЕКТУРА И ПРОЕКТИРОВАНИЕ

### 1.1 Анализ текущей системы логирования

#### Текущее состояние
| Лист | Назначение | Статус | Действие |
|------|-----------|--------|---------|
| `SESSION_LOG` ("Логи") | Не используется активно | ❌ УДАЛИТЬ | Удалить из конфига и из таблицы |
| `LOG` ("Журнал синхро") | Логирование синхронизации | ✅ ОСТАВИТЬ | Источник данных для Дашборда |
| `LOG_DEBUG` ("Журнал логов") | Полный технический лог | ✅ ОСТАВИТЬ | Источник данных для Дашборда |

#### Текущая структура логов
```javascript
// Текущая функция в 01Config.js:
function logWithEmoji(message, level, emoji, functionName, details)
// Параметры: сообщение, уровень, эмодзи, имя функции, детали
```

#### Проблемы текущей системы
- [ ] Нет категоризации логов (Article, Sync, Delete и т.д.)
- [ ] Нет статуса операции (START, PROGRESS, SUCCESS, ERROR)
- [ ] Сложно искать конкретную операцию
- [ ] Нет визуализации на основе типа операции

### 4.1 Новая сигнатура функции logWithEmoji()

#### Новая версия (обновить в 01Config.js)
```javascript
/**
 * Унифицированное логирование с категоризацией
 * @param {string} message - Сообщение на русском
 * @param {string} level - DEBUG | INFO | WARN | ERROR
 * @param {string} category - Article | Sync | Delete | Network | Price | etc.
 * @param {string} status - START | PROGRESS | SUCCESS | WARNING | ERROR
 * @param {string} functionName - Имя функции
 * @param {object} variables - Входные параметры {param1: val1, param2: val2}
 * @param {object} result - Результаты {rows: 150, duration: 2450}
 */
function logWithEmoji(message, level, category, status, functionName, variables, result) {
  // Проверка уровня логирования
  const currentLevel = SETTINGS.CURRENT_LOG_LEVEL;
  const messageLevelValue = SETTINGS.LOG_LEVELS[level] || 1;
  if (messageLevelValue < currentLevel) return;

  // Определяем эмодзи по статусу (если не передан)
  const emoji = _getEmojiForStatus(status) || _getEmojiForCategory(category);

  // Формируем полное сообщение
  const fullMessage = emoji + ' ' + message;

  // Временная метка
  const timestamp = Utilities.formatDate(
    new Date(),
    SETTINGS.TIMEZONE,
    'yyyy-MM-dd HH:mm:ss'
  );

  // 1. Лог в консоль
  Logger.log(`[${timestamp}] ${level}: ${fullMessage}`);

  // 2. Лог в Google Sheets (LOG_DEBUG)
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) return;

    const debugSheet = ss.getSheetByName(SHEETS.LOG_DEBUG);
    if (debugSheet) {
      debugSheet.appendRow([
        timestamp,           // A: Время
        category,            // B: Категория
        status,              // C: Статус
        level,               // D: Уровень
        fullMessage,         // E: Сообщение
        functionName,        // F: Функция
        JSON.stringify(variables || {}),  // G: Входные параметры
        JSON.stringify(result || {})       // H: Результаты
      ]);
    }
  } catch (e) {
    console.warn('Ошибка логирования:', e.message);
  }
}

function _getEmojiForStatus(status) {
  const statusEmojis = {
    'START': '🔵',
    'PROGRESS': '🟡',
    'SUCCESS': '🟢',
    'WARNING': '🟠',
    'ERROR': '🔴'
  };
  return statusEmojis[status] || '';
}

function _getEmojiForCategory(category) {
  const categoryEmojis = {
    'Article': '📌',
    'Sync': '🔄',
    'Delete': '🗑️',
    'Network': '🌐',
    'Price': '💰',
    'Certificate': '📜',
    'Invoice': '📄',

    'Cache': '💾',
    'Lock': '🔒',
    'Settings': '⚙️'
  };
  return categoryEmojis[category] || 'ℹ️';
}

function addArticleManually() {
  const ui = SpreadsheetApp.getUi();
  const ssId = SpreadsheetApp.getActiveSpreadsheet().getId();

  try {
    // ✅ ШАГ 1: Логируем начало
    logWithEmoji(
      "Запуск функции добавления артикула",
      "INFO",
      "Article",
      "START",
      "addArticleManually",
      { spreadsheet_id: ssId }
    );

    // ✅ ШАГ 2: Запрашиваем инпут
    logWithEmoji(
      "Ожидание ввода продукта",
      "DEBUG",
      "Article",
      "PROGRESS",
      "addArticleManually"
    );

    const prompt = ui.prompt("Добавить артикул", "Введите название продукта...", ui.ButtonSet.OK_CANCEL);
    if (prompt.getSelectedButton() !== ui.Button.OK) {
      logWithEmoji(
        "Пользователь отменил ввод",
        "INFO",
        "Article",
        "WARNING",
        "addArticleManually",
        null,
        { cancelled: true }
      );
      return;
    }
    const productName = prompt.getResponseText().trim();

    // ✅ ШАГ 3: Логируем попытку создания
    logWithEmoji(
      "Отправляем запрос на сервер создания артикула",
      "DEBUG",
      "Article",
      "PROGRESS",
      "addArticleManually",
      { productName: productName }
    );

    // ✅ ШАГ 4: Вызываем сервер
    const res = Lib.callServer("/sync/add-article", {
      article: "",
      spreadsheet_id: ssId,
      project: "UNKNOWN"
    });

    const newArticle = res && res.article ? res.article : null;

    // ✅ ШАГ 5: Логируем успех
    logWithEmoji(
      `Артикул ${newArticle} создан на сервере`,
      "INFO",
      "Article",
      "SUCCESS",
      "addArticleManually",
      { product_name: productName },
      { article_id: newArticle, status: res.status }
    );

    ui.alert(`✅ Артикул создан: ${newArticle}`);

  } catch (e) {
    // ✅ ШАГ 6: Логируем ошибку
    logWithEmoji(
      `Ошибка при создании артикула: ${e.message}`,
      "ERROR",
      "Article",
      "ERROR",
      "addArticleManually",
      null,
      { error: e.message, stack: e.stack }
    );

    ui.alert(`❌ Ошибка: ${e.message}`);
    throw e;
  }
}
```
